' Copies values from an ADO recordset into the unbound text boxes and combo boxes of a
' form. Each control's Tag holds the name of the field to read.
'
' Takes the form object, not its name, so it works with form instances: Forms("Name")
' reaches only the first instance when several share one name.
'
' Usage   : Set frm = New Form_Customer
'           PopulateUnboundControls frm, rs      ' Populate before Add, while it is hidden
'           FormInstances.Add frm, "Customer:" & lngID
'
' Requires: Reference to Microsoft ActiveX Data Objects.
Public Function PopulateUnboundControls( _
    ByVal frm As Access.Form, _
    DataRecordset As ADODB.Recordset)

    ' ByVal matters: clearing frm in the cleanup below must not clear the caller's variable.

    On Error GoTo ErrorHandler

    Dim ctl As Control
    Dim fld As ADODB.Field

    If DataRecordset.BOF And DataRecordset.EOF Then Exit Function

    For Each ctl In frm.Controls

        Select Case ctl.ControlType
            Case acTextBox, acComboBox

                If Len(ctl.Tag) > 0 Then

                    Set fld = Nothing

                    On Error Resume Next
                    Set fld = DataRecordset.Fields(ctl.Tag)
                    On Error GoTo ErrorHandler

                    If Not fld Is Nothing Then
                        ctl.Value = fld.Value
                    Else
                        Debug.Print "Field not found: " & ctl.Tag & _
                                    " (" & ctl.Name & ")"
                    End If

                End If

        End Select

    Next ctl

ExitHere:
    Set fld = Nothing
    Set ctl = Nothing
    Set frm = Nothing
    Exit Function

ErrorHandler:
    Debug.Print "PopulateUnboundControls error " & Err.Number & ": " & Err.Description

    GoTo ExitHere

End Function
